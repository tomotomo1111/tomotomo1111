class ScoreConfirmation2{
	double Eng, Math;
	public ScoreConfirmation2(double score1, double score2) {
		Eng = score1;
		Math = score2;

class Calculate2{
	static final double rate = 0.7;
	public static double ScoreCorrection(double score) {
		return score*rate;
	}
