class SeisekiData2{
	int Eng, Math, Lite;
	
	public SeisekiData2(int score1, int score2, int score3) {
		Eng = score1;
		Math = score2;
		Lite = score3;
	}
	
	public double AveScore() {
		return (Eng + Math + Lite) / 3;
	}
}

class SeisekiCalculate2{
	
	public static double[] MaxAveScore(SeisekiData2[] array) {
		double[] answer = new double[2];
		answer[0] = array[0].AveScore();
		for(int i=0; i<array.length; i++) {
			if(answer[0] <= array[i].AveScore()){
				answer[0] = array[i].AveScore();
				answer[1] = i + 1;
			}
			for(double db: answer) {
				System.out.println(db);
			}
		}
		return answer;
	}
}
