public class Enshu2Monda4{
	
	public static void main(String[] args) {
		int[] data = new int[5];
		for(int i = 0; i < data.length; i++) {
			data[i] = randomGenerate();
		}
		
		HeightData2 mydata = new HeightData2(data);
		mydata.MaxHeight();
		mydata.MinHeight();
	}
	
	public static int randomGenerate() {
		int rand;
		do {
			rand = (int)(Math.random() * 190);
		}while(rand <140);
		return rand;
	}
}

class HeightData2 {
	int[] height;
	
	public HeightData2(int[] input) {
		height = input;
	}
	
	void MaxHeight() {
		System.out.println(Math.max(height));
	}
	
	void MinHeight() {
		System.out.println(Math.min(height));
	}
}
