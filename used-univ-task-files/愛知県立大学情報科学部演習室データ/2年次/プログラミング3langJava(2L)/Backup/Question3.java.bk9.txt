	public static void updateProfit(String name, int uriage) {
		AccountPlus account = getCompany(name);
		if(uriage >= 0 && uriage <= 500) {
			account.uriage = uriage;
		}else if(uriage > 500 && uriage <= 1100) {

	public static AccountPlus getCompany(String name) {
		return companyArray[getCompanyIndexFromName(name)];
	}
	
	public static int getCompanyIndexFromName(String name) {
		int i = 0;
		while(!name.equals(companyArray[i].name)) {
			i++;
		}
		return i;
	}
	
	public static String calculateRank(String name) {
		int sum=0;
		for(int i=0; i<companyArray.length;i++) {
			sum += companyArray[i].getHokan();
		}
		double ave = sum / (double)companyArray.length;
		int hokan = companyArray[getCompanyIndexFromName(name)].getHokan();
		String answer = null;
		if(hokan >= ave + 100) {
			answer = "A";
		} else if(hokan > ave && hokan < ave + 100) {
			answer = "B";
		} else if(hokan >= ave -100 && hokan <= ave) {
			answer = "C";
		} else {
			answer = "D";
		}
		return answer;
	}

class Account {
	String name;
	int uriage;
	int rieki;
	
	public Account(String name, int uriage, int rieki) {
		this.name = name;
		this.uriage = uriage;
		this.rieki = rieki;
	}
}

class AccountPlus extends Account{
	private int hokan;
	
	public AccountPlus(String name, int uriage, int rieki, int hokan) {
		super(name, uriage, rieki);
		this.hokan = hokan;
	}
	
	public void setHokan(int hokan) {
		this.hokan = hokan;
	}
	
	public int getHokan() {
		return hokan;
	}
}
