public class Question2of4 {
	public static void main(String[] args) {
		int[] a = {1,2,3,,2,2,2,4,5};
		
		System.out.println(averageBelow(a));
	}
	
	public static int averageBelow(int[] x) {
		int sum = 0;
		double average = 0;
		int count = 0;
		
		for(int i = 0;i<x.length;i++){
			
			sum +=x[i];
			
		}
		
		average = sum / x.length;
		
		for(int i = 0;i<x.length;i++){
			
			if(x[i] <= average) count++;
			
		}
		return count;
	}
}
