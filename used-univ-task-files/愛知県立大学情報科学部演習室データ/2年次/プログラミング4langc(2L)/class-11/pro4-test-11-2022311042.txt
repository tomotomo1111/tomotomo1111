5C2 = factorial(5) / (factorial(5 - 2) * factorial(2))
    = factorial(5) / (factorial(3) * factorial(2))
    = (5 * factorial(4)) / (3 * factorial(2) * 2 * factorial(1))
    = (5 * 4 * factorial(3)) / (3 * 2 * factorial(1) * 2 * 1 * factorial(0))
    = (5 * 4 * 3 * factorial(2)) / (3 * 2 * 1 * factorial(0) * 2 * 1 * 1)
    = (5 * 4 * 3 * 2 * factorial(1)) / (3 * 2 * 1 * 1 * 2 * 1 * 1)
    = (5 * 4 * 3 * 2 * 1 * factorial(0)) / (3 * 2 * 1 * 1 * 2 * 1 * 1)
    = (5 * 4 * 3 * 2 * 1 * 1) / (3 * 2 * 1 * 1 * 2 * 1 * 1)
    = 120 / 12
    = 10
