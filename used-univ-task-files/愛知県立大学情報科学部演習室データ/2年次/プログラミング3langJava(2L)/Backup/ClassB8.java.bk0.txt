class ScoreData8{
	double Eng, Math, Lite, Total;
	private static final double rate = 0.9;
	public ScoreData8(double score1, double score2, double score3) {
		Eng = score1 * rate;
		Math = score2 * rate;
		Lite = score3 * rate;
		Total = Eng + Math + Lite;
	}
	
	public double ScoreSum() {
		return Total;
	}
}

class Calculate4{
	public static double ScoreDifference(double score1, double score2) {
		return Math.abs(score1 - score2);
	}
}
