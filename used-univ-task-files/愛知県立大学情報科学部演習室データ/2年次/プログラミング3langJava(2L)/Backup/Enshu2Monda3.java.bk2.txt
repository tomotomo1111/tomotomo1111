public class Enshu2Monda3 {
	public static void main(String[] args) {
		int former, latter;
		
		former = 180;
		latter = 170;
		
		HeightData data = new HeightData(former, latter);
		data.HeightComparison();
	}
}

class HeightData{
	int height1, height2;
	
	public HeightData(int height1, int height2) {
		
		this.height1 = height1;
		this.height2 = height2;
	}
	
	public void HeightComparison() {
		if(height1 > height2) {
			System.out.println("1番目の身長が2番目の身長よりも高い");
		} else if (height1 < height2) {
			System.out.println("2番目の身長が1番目の身長よりも高い");
		} else {
			System.out.println("高さが同じです");
		}
	}
}
