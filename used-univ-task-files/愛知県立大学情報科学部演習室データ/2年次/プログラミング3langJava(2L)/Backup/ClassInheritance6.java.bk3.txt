class InformationScience extends RobotCourse {
	public int Literacy;
	public int Programming;
	public int Informatics;
	public InformationScience(int score1, int score2, int score3, int score4, int score5, int score6) {
		super(score1, score2, score3);
		Literacy = score4;
		Programming = score5;
		Informatics = score6;
	}
	public boolean JudgementSubject() {
		if(Literacy < 60 || Programming < 60 || Informatics < 60) {
			return false;
		}else {
			return true;
		}
	}
	
	public boolean JudgementUndergraduate() {
		if(JudgementCourse() == false || JudgementSubject() == false) {
			return false;
		}else{
			return true;
		}
	}
}

class RobotCourse{
	public int control;
	public int mechanics;
	public int AI;
	public RobotCource(int score1, int score2, int score3) {
		control = score1;
		mechanics = score2;
		AI = score3;
	}
	public boolean JudgementCource() {
		if(control < 60 || mechanics < 60 || AI < 60) {
			return false;
		}else{
			return true;
		}
	}
}
