class ScoreData6{
	double Eng, Math, Lite;
	public ScoreData6(double score1, double score2, double score3){
	Eng = score1;
	Math = score2;
	Lite = score3;
	}

class Calculate3{
	static final double rate = 0.8;
	public static double ScoreCorrection(double score) {
		return score * rate;
	}
}
