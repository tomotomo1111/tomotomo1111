class Grades9 {
	int Eng, Math;
	
	public Grades9(int score1, int score2) {
		Eng = score1;
		Math = score2;
	}
	
	public int SumScore() {
		return Eng + Math;
	}
	
	public double AveScore() {
		return (Eng + Math) / 2;
	}
}
