public class Question1of4 {
	public static void main(String[] args) {
		int x = 12;
		int y = 15;
		
		if(x > y) {
			System.out.println(euc(x, y));
		} else {
			System.out.println(euc(y, x));
		}
	}
	
	private static int euc(int a, int b) {
		int r ;
		do{
			r = a % b;
			if(r != 0) {
				a = b;
				b = r;
			}
		} while(r != 0);
		return b;
	}
}
